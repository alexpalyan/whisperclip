```swift
func speakerPaletteColor(_ speaker: Speaker) -> Color {
    let palette: [Color] = [
        .blue, .purple, .teal, .orange, .pink, .green, .yellow, .red, .indigo,
    ]
    if speaker == .unknown { return .gray }
    let index = speaker.colorIndex
    return palette[min(index, palette.count - 1)]
}
```
