### Edit 4 — Add new private methods in a VAD Coordination MARK section

Add these methods in a new `// MARK: - VAD Coordination` section before `// MARK: - Helpers`:

```swift
// MARK: - VAD Coordination

/// Called from onMicrophoneLevel (per D-01). Feeds level to VADStateMachine.
/// Creates a pending segment on .startSpeech; starts ghost cleanup task on .endSpeech.
private func handleMicLevelUpdate(db: Float) async {
    guard let vad = micVAD, isRecording else { return }
    guard let event = await vad.update(levelDB: db) else { return }

    switch event {
    case .startSpeech:
        guard pendingMicSegmentId == nil else { return }  // Already have one — no duplicate
        let now = Date().timeIntervalSince(startTime ?? Date())
        let seg = createPendingSegment(speaker: .me, startTime: now)
        pendingMicSegmentId = seg.id
        Logger.log("VAD mic: created pending segment \(seg.id) at \(String(format: "%.2f", now))s", log: Logger.general)

    case .endSpeech:
        // Start ghost-cleanup task. If ASR fills the segment before 2s, task is cancelled first.
        startGhostCleanupTask(channel: .mic)
    }
}

/// Called from onSpeechDetected (per D-02). Creates system pending segment guarded by diarizer.
private func handleSystemSpeechDetected(startTime: TimeInterval, speakerLabel: String) {
    guard pendingSystemSegmentId == nil else { return }  // Already have one — diarizer fires repeatedly
    let speaker = Speaker(displayName: speakerLabel)
    let seg = createPendingSegment(speaker: speaker, startTime: startTime)
    pendingSystemSegmentId = seg.id
    Logger.log("VAD system: created pending segment \(seg.id) for \(speakerLabel) at \(String(format: "%.2f", startTime))s", log: Logger.general)
    startGhostCleanupTask(channel: .system)
}

/// Unified factory: creates a pending segment, calls transcriptCallback, increments segmentCount.
@discardableResult
private func createPendingSegment(speaker: Speaker, startTime: TimeInterval) -> MeetingSegment {
    let segment = MeetingSegment(
        speaker: speaker,
        text: "",
        startTime: startTime,
        endTime: startTime,   // Zero-duration until ASR fills it
        confidence: 0,
        isPending: true
    )
    segmentCount += 1
    transcriptCallback?(segment)
    return segment
}

/// Starts a 2-second ghost cleanup task for the given channel (per D-05).
/// Any previously running task for that channel is cancelled first (reschedule on re-trigger).
private func startGhostCleanupTask(channel: AudioChannel) {
    switch channel {
    case .mic:
        ghostMicCleanupTask?.cancel()
        ghostMicCleanupTask = Task { @MainActor [weak self] in
            try? await Task.sleep(for: .seconds(2))
            guard !Task.isCancelled, let self = self else { return }
            self.cleanupGhostSegment(channel: .mic)
        }
    case .system:
        ghostSystemCleanupTask?.cancel()
        ghostSystemCleanupTask = Task { @MainActor [weak self] in
            try? await Task.sleep(for: .seconds(2))
            guard !Task.isCancelled, let self = self else { return }
            self.cleanupGhostSegment(channel: .system)
        }
    }
}

/// Removes the pending segment for the given channel if it is still empty (ghost).
/// Called 2s after speech activity ends. Skips if segment has already received text (per D-10).
private func cleanupGhostSegment(channel: AudioChannel) {
    switch channel {
    case .mic:
        guard let id = pendingMicSegmentId else { return }
        if let seg = MeetingSession.shared.liveTranscript.first(where: { $0.id == id }),
           seg.text.isEmpty {
            Logger.log("VAD mic: removing ghost segment \(id)", log: Logger.general)
            MeetingSession.shared.removeSegment(id: id)
        }
        pendingMicSegmentId = nil

    case .system:
        guard let id = pendingSystemSegmentId else { return }
        if let seg = MeetingSession.shared.liveTranscript.first(where: { $0.id == id }),
           seg.text.isEmpty {
            Logger.log("VAD system: removing ghost segment \(id)", log: Logger.general)
            MeetingSession.shared.removeSegment(id: id)
        }
        pendingSystemSegmentId = nil
    }
}
```
