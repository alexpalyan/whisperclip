Create `Sources/VADStateMachine.swift`:

```swift
import Foundation
import Accelerate

/// Events emitted by VADStateMachine.
enum VADEvent: Equatable {
    case startSpeech
    case endSpeech
}

/// Pure-logic Voice Activity Detection state machine.
/// Call `update(levelDB:)` from the audio-level callback; returns non-nil only on state transitions.
///
/// Thread-safety: actor-isolated. Callers must `await`.
actor VADStateMachine {

    // MARK: - Configuration

    /// RMS threshold in dBFS. Signals above this value count as "active". Default: -45dB.
    let thresholdDB: Float

    /// Duration the signal must remain above threshold before emitting .startSpeech.
    let onsetDuration: Duration

    /// Duration the signal must remain below threshold before emitting .endSpeech.
    let offsetDuration: Duration

    // MARK: - State

    private var isActive: Bool = false
    /// When did we first exceed (or drop below) the threshold in the current run?
    private var onsetStart: ContinuousClock.Instant? = nil
    private var offsetStart: ContinuousClock.Instant? = nil

    // MARK: - Init

    init(
        thresholdDB: Float = -45,
        onsetDuration: Duration = .milliseconds(200),
        offsetDuration: Duration = .milliseconds(500)
    ) {
        self.thresholdDB = thresholdDB
        self.onsetDuration = onsetDuration
        self.offsetDuration = offsetDuration
    }

    // MARK: - Public Interface

    /// Feed the current RMS level (dBFS). Returns a VADEvent on state change, nil otherwise.
    /// Designed for high-frequency calls (~10Hz from processMicrophoneBuffer).
    func update(levelDB: Float) -> VADEvent? {
        let now = ContinuousClock.now

        if levelDB > thresholdDB {
            // Signal is above threshold — potential speech onset
            offsetStart = nil  // Reset silence run

            if !isActive {
                if let start = onsetStart {
                    // Evaluate accumulated above-threshold run
                    if now - start >= onsetDuration {
                        isActive = true
                        onsetStart = nil
                        Logger.log(
                            "VADStateMachine: speech started (level \(String(format: "%.1f", levelDB))dB)",
                            log: Logger.general
                        )
                        return .startSpeech
                    }
                } else {
                    // First frame above threshold — start timing the onset run
                    onsetStart = now
                }
            }
            // Already active — nothing to do

        } else {
            // Signal is below threshold — potential speech offset
            onsetStart = nil  // Reset onset run

            if isActive {
                if let start = offsetStart {
                    if now - start >= offsetDuration {
                        isActive = false
                        offsetStart = nil
                        Logger.log(
                            "VADStateMachine: speech ended (level \(String(format: "%.1f", levelDB))dB)",
                            log: Logger.general
                        )
                        return .endSpeech
                    }
                } else {
                    offsetStart = now
                }
            }
            // Not active — nothing to do
        }

        return nil
    }

    /// Compute RMS level in dBFS from raw PCM samples using vDSP (zero-allocation).
    /// Returns -160 for empty input.
    nonisolated static func rmsDB(samples: [Float]) -> Float {
        guard !samples.isEmpty else { return -160 }
        var rms: Float = 0
        vDSP_rmsqv(samples, 1, &rms, vDSP_Length(samples.count))
        return 20 * log10(max(rms, 1e-7))
    }
}
```

func testVAD01OnsetReturnsSpeechStartedAfter200ms() async throws {
    // Use a tiny onset to avoid real-time waiting in tests
    let vad = VADStateMachine(thresholdDB: -45, onsetDuration: .milliseconds(1), offsetDuration: .milliseconds(500))
    // First call starts the onset clock
    let result1 = await vad.update(levelDB: -30)
    XCTAssertNil(result1, "First above-threshold call starts timing, should not emit yet")
    // Brief pause so the 1ms onset elapses
    try await Task.sleep(for: .milliseconds(5))
    // Second call evaluates the elapsed onset
    let result2 = await vad.update(levelDB: -30)
    XCTAssertEqual(result2, .startSpeech, "Should emit .startSpeech after onset duration elapsed")
}

func testVAD01SingleFrameAboveThresholdDoesNotTrigger() async throws {
    let vad = VADStateMachine(thresholdDB: -45, onsetDuration: .milliseconds(200), offsetDuration: .milliseconds(500))
    let result = await vad.update(levelDB: -30)
    XCTAssertNil(result, "Single above-threshold frame must not emit .startSpeech — debounce not elapsed")
}

func testVAD01NoDuplicateStartWhenAlreadyActive() async throws {
    let vad = VADStateMachine(thresholdDB: -45, onsetDuration: .milliseconds(1), offsetDuration: .milliseconds(500))
    _ = await vad.update(levelDB: -30)
    try await Task.sleep(for: .milliseconds(5))
    let first = await vad.update(levelDB: -30)
    XCTAssertEqual(first, .startSpeech, "First crossing should emit .startSpeech")
    let second = await vad.update(levelDB: -30)
    XCTAssertNil(second, "Already active — must not emit .startSpeech again")
}
