import Foundation
import WhisperKit

class VoiceToTextModel: VoiceToTextProtocol {
    static let shared = VoiceToTextModel()
    private var pipe: WhisperKit?

    private init() { self.pipe = nil }

    func load() async throws {
        if self.pipe == nil {
            self.pipe = try await LocalWhisperKit.loadModel(modelRepo: CurrentSTTModelRepo, modelName: CurrentSTTModelName)
        }
        if self.pipe == nil {
            throw NSError(domain: "Transcriber", code: -1,
                          userInfo: [NSLocalizedDescriptionKey: "Failed to load voice-to-text model. Please try again later."])
        }
    }

    func process(filepath: String) async throws -> String {
        try await load()
        let language = SettingsStore.shared.language
        let languageCode = language == "auto" ? nil : language
        if GenericHelper.logSensitiveData() {
            Logger.log("Sending transcription query to WhisperKit with lang: \(languageCode ?? "auto")", log: Logger.general)
        }
        let ts = TimeSpenter()
        let results = try await pipe!.transcribe(
            audioPath: filepath,
            decodeOptions: DecodingOptions(
                language: languageCode,
                usePrefillPrompt: true,
                detectLanguage: language == "auto"
            )
        )
        let transcription = results.first?.text
        if GenericHelper.logSensitiveData() {
            Logger.log("Received transcription result from WhisperKit in \(ts.getDelay()) us", log: Logger.general)
        }
        if transcription == nil {
            Logger.log("No transcription result received", log: Logger.general, type: .error)
            throw NSError(domain: "Transcriber", code: -1,
                          userInfo: [NSLocalizedDescriptionKey: "No transcription result received"])
        }
        return transcription!
    }
}
```

import Foundation

/// Protocol defining the interface for audio transcription.
protocol VoiceToTextProtocol {
    /// Transcribe an audio file to text.
    /// - Parameter filepath: The path to the audio file (WAV format).
    /// - Returns: The transcribed text.
    /// - Throws: TranscriptionError if the transcription fails.
    func process(filepath: String) async throws -> String

    /// Transcribe an audio file to text with streaming token callback.
    /// - Parameters:
    ///   - filepath: Path to the audio file (WAV format).
    ///   - onToken: Called on the MainActor each time new partial text is available.
    ///              Receives the cumulative text decoded so far.
    /// - Returns: The final complete transcription string.
    /// - Throws: TranscriptionError if the transcription fails.
    func processStream(
        filepath: String,
        onToken: @escaping @MainActor (String) -> Void
    ) async throws -> String
}

extension VoiceToTextProtocol {
    /// Default implementation: delegates to `process` without streaming.
    /// Conformers that support real-time streaming (e.g. WhisperKit) should override this.
    /// Parakeet uses this default — onToken fires once with the final result.
    func processStream(
        filepath: String,
        onToken: @escaping @MainActor (String) -> Void
    ) async throws -> String {
        let result = try await process(filepath: filepath)
        await MainActor.run { onToken(result) }
        return result
    }
}

func processStream(
    filepath: String,
    onToken: @escaping @MainActor (String) -> Void
) async throws -> String {
    try await load()

    let language = SettingsStore.shared.language
    let languageCode = language == "auto" ? nil : language

    if GenericHelper.logSensitiveData() {
        Logger.log("WhisperKit streaming transcription started for lang: \(languageCode ?? "auto")", log: Logger.general)
    }
    let ts = TimeSpenter()

    let results = try await pipe!.transcribe(
        audioPath: filepath,
        decodeOptions: DecodingOptions(
            language: languageCode,
            usePrefillPrompt: true,
            detectLanguage: language == "auto"
        )
    ) { progress in
        // progress.segments contains cumulative decoded segments for the current audio
        let currentText = progress.segments.map { $0.text }.joined()
        Task { @MainActor in
            onToken(currentText)
        }
        return true  // Always continue — false would abort decoding mid-stream
    }

    let transcription = results.first?.text ?? ""
    if GenericHelper.logSensitiveData() {
        Logger.log("WhisperKit streaming done in \(ts.getDelay()) us", log: Logger.general)
    }

    if transcription.isEmpty {
        Logger.log("No transcription result received from WhisperKit streaming", log: Logger.general, type: .error)
        throw NSError(domain: "Transcriber", code: -1,
                      userInfo: [NSLocalizedDescriptionKey: "No transcription result received"])
    }

    return transcription
}
```
