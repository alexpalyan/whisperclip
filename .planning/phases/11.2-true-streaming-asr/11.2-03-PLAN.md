**Step 2 — Add enqueueFragment** (new method, after existing enqueue overloads):

```swift
func enqueueFragment(
    source: AudioSource,
    samples: [Float],
    startTime: TimeInterval,
    processor: @escaping @MainActor (AudioSource, [Float], TimeInterval) async -> Void
) async {
    let request = FragmentRequest(source: source, samples: samples,
                                  startTime: startTime, processor: processor)
    if source == .microphone {
        micFragments.append(request)
    } else {
        // Backpressure: drop oldest system fragments if queue is too deep
        while totalQueuedDuration(for: .system) + Double(samples.count) / Double(fragmentSampleRate) > maxQueuedSeconds,
              !systemFragments.isEmpty {
            Logger.log(
                "TranscriptionQueue: dropping oldest system fragment (backpressure, queue > \(maxQueuedSeconds)s)",
                log: Logger.general,
                type: .error
            )
            systemFragments.removeFirst()
        }
        systemFragments.append(request)
    }

    guard !isProcessing else { return }
    await processNextFragment()
}
```

**Step 3 — Add totalQueuedDuration and processNextFragment** (new private methods):

```swift
func totalQueuedDuration(for source: AudioSource) -> TimeInterval {
    let lane = source == .microphone ? micFragments : systemFragments
    return lane.reduce(0.0) { $0 + Double($1.samples.count) / Double(fragmentSampleRate) }
}

private func processNextFragment() async {
    // Mic has absolute priority
    let next: FragmentRequest?
    if !micFragments.isEmpty {
        next = micFragments.removeFirst()
    } else if !systemFragments.isEmpty {
        next = systemFragments.removeFirst()
    } else {
        return
    }
    guard let fragment = next else { return }
    isProcessing = true
    await fragment.processor(fragment.source, fragment.samples, fragment.startTime)
    isProcessing = false
    // Check existing pending work first, then fragment lanes
    if !pendingRequests.isEmpty || !pendingBufferRequests.isEmpty {
        await processNext()
    } else {
        await processNextFragment()
    }
}
```

private func makePCMBuffer(from samples: [Float]) -> AVAudioPCMBuffer? {
    guard let format = AVAudioFormat(
        commonFormat: .pcmFormatFloat32,
        sampleRate: 16000,
        channels: 1,
        interleaved: false
    ) else { return nil }
    guard let buffer = AVAudioPCMBuffer(
        pcmFormat: format,
        frameCapacity: AVAudioFrameCount(samples.count)
    ) else { return nil }
    buffer.frameLength = AVAudioFrameCount(samples.count)
    samples.withUnsafeBufferPointer { ptr in
        buffer.floatChannelData?[0].update(from: ptr.baseAddress!, count: samples.count)
    }
    return buffer
}

**Step 4 — Implement feedFragment**:

```swift
func feedFragment(
    _ samples: [Float],
    source: AudioSource,
    onToken: @escaping @MainActor @Sendable (String) -> Void
) async throws {
    let session: StreamingEouAsrManager?
    switch source {
    case .microphone: session = streamingSessionMic
    case .system:     session = streamingSessionSystem
    }
    guard let activeSession = session else {
        Logger.log("feedFragment: no Parakeet session for \(source) — call startStreamingSession first",
                   log: Logger.general, type: .error)
        return
    }
    guard !samples.isEmpty else { return }

    if GenericHelper.logSensitiveData() {
        Logger.log("feedFragment: feeding \(samples.count) samples to Parakeet \(source)", log: Logger.general)
    }

    // Re-register the partial callback with the current onToken closure for this fragment.
    // Callbacks are @Sendable but NOT @MainActor — dispatch manually.
    activeSession.setPartialCallback { [onToken] text in
        guard !text.isEmpty else { return }
        Task { @MainActor in onToken(text) }
    }

    guard let pcmBuffer = makePCMBuffer(from: samples) else {
        Logger.log("feedFragment: failed to create AVAudioPCMBuffer for \(source)",
                   log: Logger.general, type: .error)
        return
    }

    // process(audioBuffer:) accumulates internally, fires partial+EOU callbacks automatically.
    // Return value is "" — partial text arrives via callback above.
    _ = try await activeSession.process(audioBuffer: pcmBuffer)
}
```
