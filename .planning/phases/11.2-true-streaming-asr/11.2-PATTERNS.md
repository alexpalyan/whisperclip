extension VoiceToTextProtocol {

        filepath: String,
        onEvent: @escaping @MainActor @Sendable (StreamingTranscriptionEvent) -> Void,
        onToken: @escaping @MainActor @Sendable (String) -> Void

func enqueue(
    source: AudioSource,
    samples: [Float],
    startTime: TimeInterval,
    processor: @escaping @MainActor (AudioSource, [Float], TimeInterval) async -> Void
) async {
    if isProcessing {
        pendingRequests.append((source, samples, startTime, processor))
        return
    }
    isProcessing = true
    await processor(source, samples, startTime)
    isProcessing = false
    await processNext()
}
