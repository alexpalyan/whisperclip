**Step 1 — Add module-internal free functions** (outside the class, at file scope, after the closing `}` of VoiceToTextModel):

```swift
// MARK: - Streaming Helpers (module-internal, tested by StreamingTests)

/// Returns the longest common prefix of `a` and `b`, truncated to the last
/// complete word boundary (space or string end). This is the "committed" stable
/// text from two consecutive decode iterations (LocalAgreement / LCP policy, D-01).
func calculateLCP(_ a: String, _ b: String) -> String {
    guard !a.isEmpty, !b.isEmpty else { return "" }
    var commonEnd = a.startIndex
    var ai = a.startIndex
    var bi = b.startIndex
    while ai < a.endIndex, bi < b.endIndex, a[ai] == b[bi] {
        if a[ai] == " " {
            commonEnd = a.index(after: ai)  // track last space boundary
        }
        ai = a.index(after: ai)
        bi = b.index(after: bi)
    }
    // If whole string matched and ends without space, return full match
    if ai == a.endIndex, bi == b.endIndex {
        return a
    }
    return String(a[..<commonEnd])
}

/// Hides the last word if it is incomplete (no trailing space or punctuation).
/// Prevents flickering half-words in the UI (Last-Word Truncation, D-09).
func applyLastWordTruncation(_ text: String) -> String {
    guard !text.isEmpty else { return "" }
    let lastChar = text.last!
    let terminators: Set<Character> = [" ", ".", ",", "!", "?", ":", ";", "…", "\n"]
    if terminators.contains(lastChar) {
        return text
    }
    // Find the last space and return everything up to and including it
    if let lastSpaceIdx = text.lastIndex(of: " ") {
        return String(text[...lastSpaceIdx])
    }
    // No space found — entire text is one unconfirmed word; hide it all
    return ""
}
```

```swift
// MARK: - Session-Based Streaming (VoiceToTextProtocol, Phase 11.2)

func startStreamingSession(source: AudioSource) async throws {
    try await load()
    guard slidingBuffers[source] == nil else {
        Logger.log("startStreamingSession: session for \(source) already active (no-op)", log: Logger.general)
        return
    }
    slidingBuffers[source] = []
    lastTranscriptions[source] = ""
    committedPrefixes[source] = ""
    Logger.log("startStreamingSession: opened session for \(source)", log: Logger.general)
}

func feedFragment(
    _ samples: [Float],
    source: AudioSource,
    onToken: @escaping @MainActor @Sendable (String) -> Void
) async throws {
    guard let pipe = pipe else {
        Logger.log("feedFragment: WhisperKit not loaded for \(source)", log: Logger.general, type: .error)
        return
    }
    guard var buffer = slidingBuffers[source] else {
        Logger.log("feedFragment: no active session for \(source) — call startStreamingSession first", log: Logger.general, type: .error)
        return
    }
    guard !samples.isEmpty else { return }

    // Append new fragment and evict oldest if window exceeds 5s
    buffer.append(contentsOf: samples)
    if buffer.count > maxSlidingWindowSamples {
        buffer.removeFirst(buffer.count - maxSlidingWindowSamples)
    }
    slidingBuffers[source] = buffer

    let language = SettingsStore.shared.language
    let languageCode = language == "auto" ? nil : language

    if GenericHelper.logSensitiveData() {
        Logger.log("feedFragment: transcribing \(buffer.count) samples for \(source)", log: Logger.general)
    }

    let results = try await pipe.transcribe(
        audioArray: buffer,
        decodeOptions: DecodingOptions(
            language: languageCode,
            usePrefillPrompt: true,
            detectLanguage: language == "auto",
            compressionRatioThreshold: 2.0,
            logProbThreshold: -0.5,
            noSpeechThreshold: 0.3
        )
    )

    guard let rawText = results.first?.text else {
        Logger.log("feedFragment: no result for \(source)", log: Logger.general, type: .debug)
        return
    }

    let currentText = rawText.trimmingCharacters(in: .whitespacesAndNewlines)
    let previousText = lastTranscriptions[source] ?? ""
    lastTranscriptions[source] = currentText

    // LCP policy: only emit the stable prefix agreed by consecutive decodes
    let lcp = calculateLCP(currentText, previousText)
    let truncated = applyLastWordTruncation(lcp)

    let alreadyCommitted = committedPrefixes[source] ?? ""
    guard truncated != alreadyCommitted, !truncated.isEmpty else { return }

    committedPrefixes[source] = truncated
    await MainActor.run {
        onToken(truncated)
    }
}

func stopStreamingSession(source: AudioSource) async throws {
    slidingBuffers.removeValue(forKey: source)
    lastTranscriptions.removeValue(forKey: source)
    committedPrefixes.removeValue(forKey: source)
    Logger.log("stopStreamingSession: closed session for \(source)", log: Logger.general)
}
```
