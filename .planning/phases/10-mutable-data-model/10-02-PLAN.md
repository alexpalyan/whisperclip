```swift
/// Splits a sample array at a sample-accurate offset. Clamps to array bounds.
/// Returns (first, second) where first contains samples before the split point.
nonisolated func splitBuffer(_ samples: [Float], atSeconds offset: Double) -> ([Float], [Float]) {
    let splitIndex = min(max(0, Int(offset * Double(sampleRate))), samples.count)
    return (Array(samples.prefix(splitIndex)), Array(samples.dropFirst(splitIndex)))
}
```

Add private `flushMicroWindow()`:
```swift
/// Flush exactly `microWindowSamples` worth of samples as a single ClosedSpeakerBuffer.
private func flushMicroWindow() {
    let windowSamples = Array(accumulatedSamples.prefix(microWindowSamples))
    let overflow = Array(accumulatedSamples.dropFirst(microWindowSamples))

    guard windowSamples.count >= minSamplesPerBuffer else {
        accumulatedSamples = overflow
        return
    }

    let label = resolveLabel(for: currentSpeakerId)
    let closed = ClosedSpeakerBuffer(
        samples: windowSamples,
        speakerLabel: label,
        startTime: bufferStartTime
    )

    Logger.log(
        "SpeakerBufferManager: micro-window flush \(label) \(windowSamples.count) samples",
        log: Logger.general
    )

    continuation?.yield(closed)
    bufferStartTime += Double(microWindowSamples) / Double(sampleRate)
    accumulatedSamples = overflow
}
```

Replace with split logic. To find the change offset: use the FIRST segment in `result.segments` whose `speakerId` matches `dominantId` — its `startSeconds` (relative to `windowStartTime`) gives the change point within the accumulated buffer:

```swift
if let current = currentSpeakerId, current != dominantId {
    // Find the timestamp of the first segment belonging to the new speaker.
    // result.segments are relative to windowStartTime; we need the offset
    // within accumulatedSamples from bufferStartTime.
    let changeOffsetFromBufferStart: Double

    if let changeSegment = result.segments.first(where: { $0.speakerId == dominantId }) {
        // changeSegment.startSeconds is relative to windowStartTime.
        // Convert to offset from bufferStartTime.
        let absoluteChangeTime = windowStartTime + Double(changeSegment.startSeconds)
        changeOffsetFromBufferStart = max(0, absoluteChangeTime - bufferStartTime)
    } else {
        // No precise timestamp available — flush at current boundary
        changeOffsetFromBufferStart = Double(accumulatedSamples.count) / Double(sampleRate)
    }

    Logger.log(
        "SpeakerBufferManager: speaker change \(current) -> \(dominantId) at +\(String(format: "%.2f", changeOffsetFromBufferStart))s",
        log: Logger.general
    )

    // Split the buffer at the speaker-change boundary
    let (firstSamples, secondSamples) = splitBuffer(accumulatedSamples, atSeconds: changeOffsetFromBufferStart)

    // Emit first segment (old speaker) if above floor
    if firstSamples.count >= minSamplesPerBuffer {
        let firstLabel = resolveLabel(for: current)
        let firstClosed = ClosedSpeakerBuffer(
            samples: firstSamples,
            speakerLabel: firstLabel,
            startTime: bufferStartTime
        )
        continuation?.yield(firstClosed)
    }

    // Prepare second segment (new speaker) as the new accumulation start
    let splitTime = bufferStartTime + changeOffsetFromBufferStart
    bufferStartTime = splitTime
    accumulatedSamples = secondSamples
}
```
