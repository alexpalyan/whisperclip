```swift
import Observation   // Add at top of file alongside existing Foundation import

@Observable
final class MeetingSegment: Identifiable, Codable {
    let id: UUID
    var speaker: Speaker
    var text: String
    let startTime: TimeInterval
    let endTime: TimeInterval
    var confidence: Float
    var isPending: Bool

    // MARK: - Init

    init(
        speaker: Speaker,
        text: String,
        startTime: TimeInterval,
        endTime: TimeInterval,
        confidence: Float = 1.0,
        isPending: Bool = false
    ) {
        self.id = UUID()
        self.speaker = speaker
        self.text = text
        self.startTime = startTime
        self.endTime = endTime
        self.confidence = confidence
        self.isPending = isPending
    }

    // MARK: - Computed helpers

    var duration: TimeInterval { endTime - startTime }

    var formattedTime: String {
        let minutes = Int(startTime) / 60
        let seconds = Int(startTime) % 60
        return String(format: "%02d:%02d", minutes, seconds)
    }

    var displayText: String { isPending ? "..." : text }
    var displaySpeaker: Speaker { isPending ? .unknown : speaker }

    // MARK: - Manual Codable (prevents @Observable macro storage leakage)

    enum CodingKeys: String, CodingKey {
        case id, speaker, text, startTime, endTime, confidence, isPending
    }

    required init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        self.id         = try c.decode(UUID.self,         forKey: .id)
        self.speaker    = try c.decode(Speaker.self,      forKey: .speaker)
        self.text       = try c.decode(String.self,       forKey: .text)
        self.startTime  = try c.decode(TimeInterval.self, forKey: .startTime)
        self.endTime    = try c.decode(TimeInterval.self, forKey: .endTime)
        self.confidence = try c.decodeIfPresent(Float.self, forKey: .confidence) ?? 1.0
        self.isPending  = try c.decodeIfPresent(Bool.self,  forKey: .isPending)  ?? false
    }

    func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        try c.encode(id,         forKey: .id)
        try c.encode(speaker,    forKey: .speaker)
        try c.encode(text,       forKey: .text)
        try c.encode(startTime,  forKey: .startTime)
        try c.encode(endTime,    forKey: .endTime)
        try c.encode(confidence, forKey: .confidence)
        try c.encode(isPending,  forKey: .isPending)
    }
}
```

**MeetingSession.swift — add three methods** after `handleNewSegment`:

```swift
// MARK: - Pending Segment API (called by MeetingRecorder and Reconciler)

/// Adds a segment in pending state to the live transcript immediately.
/// Does NOT persist — storage.addSegment is called only after ASR resolves in finalizeSegment.
@MainActor
func addPendingSegment(_ segment: MeetingSegment) {
    guard isActive else { return }
    liveTranscript.append(segment)
    liveTranscript.sort { $0.startTime < $1.startTime }
}

/// Updates an existing pending segment in-place with resolved text and speaker.
/// Called by Reconciler (Phase 13) or directly by MeetingRecorder after ASR finishes.
@MainActor
func finalizeSegment(id: UUID, text: String, speaker: Speaker) {
    guard let seg = liveTranscript.first(where: { $0.id == id }) else {
        Logger.log("MeetingSession.finalizeSegment: segment \(id) not found in liveTranscript", log: Logger.general, type: .error)
        return
    }
    withAnimation(.easeInOut(duration: 0.2)) {
        seg.text = text
        seg.speaker = speaker
        seg.isPending = false
    }
    if let meetingId = currentMeetingId {
        storage.addSegment(seg, to: meetingId)
    }
}

/// Replaces a single segment (e.g. after a speaker-boundary split) with two new segments.
/// Called by Reconciler for split case. The first replacement segment inherits the original's ID position.
@MainActor
func replaceSegment(id: UUID, with newSegments: [MeetingSegment]) {
    guard let index = liveTranscript.firstIndex(where: { $0.id == id }) else {
        Logger.log("MeetingSession.replaceSegment: segment \(id) not found", log: Logger.general, type: .error)
        return
    }
    withAnimation(.easeInOut(duration: 0.2)) {
        liveTranscript.remove(at: index)
        liveTranscript.insert(contentsOf: newSegments, at: index)
    }
    if let meetingId = currentMeetingId {
        for seg in newSegments {
            storage.addSegment(seg, to: meetingId)
        }
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| JSON persistence (UserDefaults) → app memory | Loaded JSON is decoded into @Observable instances; malformed data crashes if not handled |
| Segment mutation from background tasks → Main thread | finalizeSegment must always run on @MainActor; off-thread mutation is a data race |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-01 | Tampering | `MeetingStorage` — JSON in UserDefaults | mitigate | `decodeIfPresent` with safe defaults for `isPending` and `confidence` ensures malformed/partial JSON does not crash; tested in `testDecodesOldStructFormatWithoutError` |
| T-10-02 | Information Disclosure | `@Observable` macro storage in JSON | mitigate | Manual `CodingKeys` + `encode(to:)` explicitly excludes `_$observationRegistrar`; verified by `testEncodedJSONHasNoObservationRegistrar` |
| T-10-03 | Denial of Service | Background task mutates @Observable off main actor | mitigate | `finalizeSegment` and `replaceSegment` are `@MainActor`; callers must use `await MainActor.run {}` or `Task { @MainActor in }` — enforced by Swift compiler under `SWIFT_STRICT_CONCURRENCY=complete` |
| T-10-04 | Tampering | Split segments lose transcript text | accept | `replaceSegment(id:with:)` inserts both new segments before removing original; compiler-enforced array atomicity in single-threaded @MainActor context |
</threat_model>
