**2. Add a private method** `accumulateSystemFallback(_:atTime:)` near `processAudioChunk` (around line 385):

```swift
/// Accumulates system audio samples when no diarizer is available.
/// When the buffer reaches 5s worth of samples (80,000 at 16kHz),
/// dispatches through TranscriptionQueue as a .system chunk attributed to .other.
private func accumulateSystemFallback(_ samples: [Float], atTime time: TimeInterval) {
    if systemFallbackBuffer.isEmpty {
        systemFallbackStartTime = time
    }
    systemFallbackBuffer.append(contentsOf: samples)

    let chunkSampleCount = Int(Double(sampleRate) * systemFallbackChunkDuration)  // 80,000
    if systemFallbackBuffer.count >= chunkSampleCount {
        let chunk = systemFallbackBuffer
        let chunkStart = systemFallbackStartTime
        systemFallbackBuffer = []

        Task { [weak self] in
            guard let self = self else { return }
            await self.transcriptionQueue.enqueue(
                source: .system,
                samples: chunk,
                startTime: chunkStart,
                processor: { src, samp, start in
                    await self.processAudioChunk(source: src, samples: samp, startTime: start)
                }
            )
        }
    }
}
```

**3. Modify the `onSystemBatch` callback** in `startRecording()` (lines 154-159). Replace:

```swift
onSystemBatch: { [weak self] samples, time in
    guard let self = self else { return }
    Task {
        await self.speakerBufferManager?.onAudioBatch(samples, atTime: time)
    }
}
```

With:

```swift
onSystemBatch: { [weak self] samples, time in
    guard let self = self else { return }
    Task {
        if let manager = await self.speakerBufferManager {
            await manager.onAudioBatch(samples, atTime: time)
        } else {
            // Fallback: no diarizer available — accumulate system audio
            // and dispatch as 5s chunks attributed to .other
            await self.accumulateSystemFallback(samples, atTime: time)
        }
    }
}
```

IMPORTANT: `self.speakerBufferManager` must be read inside the Task (it is on @MainActor). The `if let` check must happen inside the `await` context, not captured in the closure. Since `MeetingRecorder` is `@MainActor`, accessing `speakerBufferManager` from within `await` (which hops to MainActor) is correct.

**Add the following test method** to `MeetingRecorderTests`:

```swift
// MARK: - GAP-08-01: System audio fallback when diarizer is unavailable

/// Verifies that when speakerBufferManager is nil (no diarizer),
/// system audio batches are accumulated and dispatched as .system chunks
/// through TranscriptionQueue, producing .other speaker attribution.
///
/// This closes the gap from 08-VERIFICATION.md: "onSystemBatch only
/// calls speakerBufferManager?.onAudioBatch(...) with no fallback branch."
func testSystemAudioFallbackWhenDiarizerUnavailable() async throws {
    // Simulate the fallback accumulation logic from MeetingRecorder.
    // When speakerBufferManager is nil, onSystemBatch should accumulate
    // samples and dispatch through TranscriptionQueue as .system source.

    let sampleRate = 16_000
    let chunkDuration = 5.0
    let chunkSampleCount = Int(Double(sampleRate) * chunkDuration)  // 80,000

    // Accumulation state (mirrors systemFallbackBuffer in MeetingRecorder)
    var fallbackBuffer: [Float] = []
    var fallbackStartTime: TimeInterval = 0
    var dispatchedChunks: [(source: AudioSource, samples: [Float], startTime: TimeInterval)] = []

    // Simulate receiving system audio batches (no speakerBufferManager)
    let batchSize = 16_000  // 1s of audio per batch
    for batchIndex in 0..<6 {
        let time = TimeInterval(batchIndex)
        let samples = Array(repeating: Float(0.1), count: batchSize)

        if fallbackBuffer.isEmpty {
            fallbackStartTime = time
        }
        fallbackBuffer.append(contentsOf: samples)

        if fallbackBuffer.count >= chunkSampleCount {
            dispatchedChunks.append((
                source: .system,
                samples: fallbackBuffer,
                startTime: fallbackStartTime
            ))
            fallbackBuffer = []
        }
    }

    // After 6 batches of 16,000 samples = 96,000 samples total
    // Should have dispatched 1 chunk of 80,000 samples with 16,000 remaining
    XCTAssertEqual(dispatchedChunks.count, 1, "Should dispatch one 5s chunk after accumulating 80,000+ samples")
    XCTAssertEqual(dispatchedChunks[0].samples.count, 80_000, "Dispatched chunk should contain 5s of audio")
    XCTAssertEqual(dispatchedChunks[0].startTime, 0.0, "First chunk should start at time 0")
    XCTAssertEqual(dispatchedChunks[0].source, .system, "Fallback chunks must be .system source")

    // Verify .system source maps to .other speaker
    let speaker = AudioSource.system.speaker
    XCTAssertEqual(speaker, .other, "System audio source must map to .other speaker for fallback attribution")

    // Verify remaining buffer is not lost (would be flushed on stopRecording)
    XCTAssertEqual(fallbackBuffer.count, 16_000, "Remaining samples should stay in buffer for flush on stop")

    // Verify the dispatched chunk passes the minimum sample threshold (2s = 32,000)
    XCTAssertGreaterThanOrEqual(
        dispatchedChunks[0].samples.count,
        sampleRate * 2,
        "Dispatched chunk must meet the 2s minimum sample threshold for ASR"
    )
}
```
